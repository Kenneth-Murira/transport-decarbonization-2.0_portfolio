# Function to convert residual matrices into tidy data frames
get_residuals_df <- function(audit_list, predictor_name) {
  
  res_matrix <- audit_list[[predictor_name]]$residuals
  
  res_df <- as.data.frame(as.table(res_matrix))
  
  colnames(res_df) <- c(
    "Predictor_Level",
    "Has_EV",
    "Std_Residual"
  )
  
  res_df$Predictor <- predictor_name
  
  return(res_df)
}
# Extract and combine standardized residuals

  mutate(
    Residual_Interpretation = case_when(
      Std_Residual > 2 ~ "More than expected",
      Std_Residual < -2 ~ "Less than expected",
      TRUE ~ "Close to expected"
    )
  )

# Plotting the Heat Map
ggplot(
  heatmap_data,
  aes(
    x = Has_EV,
    y = Predictor_Level,
    fill = Std_Residual
  )
) +
  geom_tile(
    color = "white",
    linewidth = 0.5
  ) +
  geom_text(
    aes(label = round(Std_Residual, 1)),
    size = 3
  ) +
  scale_fill_gradient2(
    low = "#D73027",
    mid = "white",
    high = "#4575B4",
    midpoint = 0,
    limits = c(
      -max(abs(heatmap_data$Std_Residual)),
      max(abs(heatmap_data$Std_Residual))
    )
  ) +
  facet_wrap(
    ~ Predictor,
    scales = "free_y",
    ncol = 1
  ) +
  theme_decarb() +
  labs(
    title = "Demographic Drivers of Early EV Adoption",
    subtitle = "Standardized Pearson Residuals: Blue = higher than expected; Red = lower than expected",
    x = "Household EV/PHEV Ownership",
    y = NULL,
    fill = "Std. Residual"
  )
